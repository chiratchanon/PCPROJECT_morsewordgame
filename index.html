Morse Word Game

บทคัดย่อ

โปรเจกต์ Morse Word Game พัฒนาขึ้นเพื่อศึกษาการแปลงเสียงเป็นรหัสมอร์สด้วย Microcontroller และการเชื่อมต่ออุปกรณ์เข้ากับเว็บผ่านอินเทอร์เน็ต ระบบใช้บอร์ด Arduino UNO R4 WiFi ร่วมกับ MAX4466 วัดความดังของเสียงและใช้ความยาวของเสียง(ms) แยกจุด-ขีด จากนั้นแปลงเป็นตัวอักษรและแสดงบนจอ LCD

เกมบนเว็บจะสุ่มคำ รับคำตอบจากแต่ละบอร์ดซึ่งแยกกันด้วย ID ตรวจความถูกต้อง และให้คะแนน ผลที่ได้คือระบบที่ผู้เล่นสะกดคำด้วยเสียงและเล่นแข่งกันได้หลายบอร์ด

เกี่ยวกับโปรเจกต์

ที่มาและเป้าหมาย

Morse Code คือ วิธีการเข้ารหัสและส่งข้อความระยะไกล โดยเปลี่ยนตัวอักษร ตัวเลข หรือเครื่องหมายวรรคตอน ให้กลายเป็นชุดสัญญาณที่มีสั้น (.) และยาว (-) ซึ่งสามารถส่งผ่านเสียงเคาะ แสงไฟ หรือการกะพริบตาได้

เป้าหมายของโปรเจกต์คือเพื่อการศึกษาเกี่ยวกับรหัสมอร์ส ทำให้การเรียนรู้รหัสมอร์ส มีความน่าสนใจมากขึ้น

วิธีทำงาน

Audio→Arduino R4 WiFi→MQTT Server→WebGame

วิธีอ่านรหัสมอร์สจากเสียง

Arduino จะวัดความดังของเสียงทุก 50ms โดยใช้ผลต่างระหว่าง MIN - MAX ถ้าเกินค่าที่กำหนดถือว่ามีเสียง แล้วใช้ความยาวของเสียงตัดสินว่าเป็นสั้น (.) หรือยาว (-)

MQTT & WebGame

เมื่อกดปุ่มส่ง บอร์ด Arduino จะส่งคำไปยังเซิร์ฟเวอร์ MQTT สาธารณะของ HiveMQ โดยใช้หัวข้อที่มี ID ของบอร์ด เพื่อให้เกมรู้ว่าคำมาจากบอร์ดไหน และตรวจสอบความถูกต้องพร้อมให้คะแนนให้กับบอร์ดแรกที่ตอบถูก

โดยที่ WebGame จะเชื่อมต่อกับเซิร์ฟเวอร์เดียวกันผ่าน WebSocket สุ่มคำให้ผู้เล่น แสดงรหัสมอร์สไว้ใต้แต่ละตัวอักษรเป็นตัวช่วย ตรวจคำที่บอร์ด Arduino ส่งมา และแสดงตารางคะแนนของแต่ละบอร์ด เกมจะเริ่มต่อเมื่อมีอย่างน้อย 1 บอร์ดออนไลน์

อุปกรณ์

อุปกรณ์หน้าที่
Arduino UNO R4 WiFiตัวประมวลผลหลักและเชื่อมต่อกับ WiFi
MAX4466ตัวรับเสียงเข้าบอร์ด
Potentiometerใช้เปิด-ปิดไมค์
LEDให้สัญญานว่ามีเสียงเข้าไมค์
Button1ลบตัวอักษร
Button2ส่งตัวอักษรเข้า MQTT
LCD I2C (16x2)แสดงผลตัวอักษรต่าง ๆ

ปัญหาที่เจอและแนวทางพัฒนาต่อ

ปัญหาการแก้ไข
ไมค์ Sensitive เกินไปเซ็ตค่า threshold และ gain ของไมค์ และเพิ่มการกรองสัญญาณให้เสียงสั้นๆ ที่เป็นสัญญาณรบกวนไม่ถูกนับ
เว็บแยกไม่ได้ว่าคำมาจากบอร์ดไหนให้แต่ละบอร์ดมี BOARD_ID
แยกจุด-ขีดโดยการ Frequency ของงเสียงพูดแต่ได้ผลลัพธ์ไม่คงที่เปลี่ยนมาใช้ความยาวของเสียงแทน ซึ่งง่ายและแม่นยำกว่า
อุปกรณ์ไม่เพียงพอImprovise อุปกรณ์เช่น ใช้ Potentiometer ในการเปิด-ปิดไมค์แทนปุ่มหรือสวิซต์

1. Arduino (morse-board.ino) 
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <WiFiS3.h>
#include <PubSubClient.h>

// ---------- Board ----------
const char* BOARD_ID = "board1";   // board2, board3... must be different on every board

// ---------- WiFi ----------
const char* WIFI_SSID     = "YOUR_WIFI_NAME";
const char* WIFI_PASSWORD = "YOUR_WIFI_PASSWORD";

// ---------- MQTT ----------
const char* MQTT_BROKER = "mqtt-dashboard.com";
const int   MQTT_PORT   = 1883;
String wordTopic;                  // pc/morse/<board id>/word
String statusTopic;                // pc/morse/<board id>/status

LiquidCrystal_I2C lcd(0x27, 16, 2);    // change to 0x3F if nothing shows
WiFiClient wifiClient;
PubSubClient client(wifiClient);

const int micPin = A0;
const int potPin = A1;                 // mic on/off knob (middle leg)
const int ledPin = 2;
const int deletePin = 3;               // delete button (other leg to GND)
const int sendPin = 4;                 // send button (other leg to GND)
const int sampleWindow = 50;           // ms
const int threshold = 100;             // adjust for your room

const int potOn  = 600;                // knob above this = mic ON
const int potOff = 400;                // knob below this = mic OFF

const unsigned long dashTime     = 400;   // longer than this = "-", shorter = "."
const unsigned long soundEndGap  = 150;   // silence this long ends a sound
const unsigned long letterGap    = 900;   // silence this long ends a letter
const unsigned long wordGap      = 2000;  // silence this long = space (Serial only)
const unsigned long debounceMs   = 50;

const bool debug = false;              // true = print loudness in Serial Monitor

const char* letters[] = {
  ".-", "-...", "-.-.", "-..", ".", "..-.", "--.", "....", "..", ".---",
  "-.-", ".-..", "--", "-.", "---", ".--.", "--.-", ".-.", "...", "-",
  "..-", "...-", ".--", "-..-", "-.--", "--.."
};
const char* digits[] = {
  "-----", ".----", "..---", "...--", "....-",
  ".....", "-....", "--...", "---..", "----."
};

bool micOn = true;
bool inSound = false;
bool wordSpaceSent = true;
unsigned long soundStart = 0, lastLoud = 0, lastEnd = 0;
unsigned long statusUntil = 0;         // when to clear the status text on row 1
String code = "";
String message = "";                   // all decoded letters (LCD shows the last 16)

struct Button {
  int pin;
  bool lastReading;
  bool state;
  unsigned long lastChange;
};
Button deleteBtn = {deletePin, HIGH, HIGH, 0};
Button sendBtn   = {sendPin,   HIGH, HIGH, 0};

// Returns true once each time the button is pressed (debounced)
bool pressedEvent(Button &b) {
  bool reading = digitalRead(b.pin);
  unsigned long now = millis();
  if (reading != b.lastReading) b.lastChange = now;
  b.lastReading = reading;

  if (now - b.lastChange > debounceMs && reading != b.state) {
    b.state = reading;
    if (b.state == LOW) return true;
  }
  return false;
}

char decodeMorse(const String &c) {
  for (int i = 0; i < 26; i++) if (c == letters[i]) return 'A' + i;
  for (int i = 0; i < 10; i++) if (c == digits[i])  return '0' + i;
  return '?';
}

// Shows the last 16 decoded letters on row 2
void updateLCD() {
  String shown = message;
  if (shown.length() > 16) shown = shown.substring(shown.length() - 16);
  lcd.setCursor(0, 1);
  lcd.print(shown);
  for (int i = shown.length(); i < 16; i++) lcd.print(' ');
}

void addToMessage(char c) {
  message += c;
  if (message.length() > 100) message.remove(0, message.length() - 100);
}

// Shows a short status on row 1 for 1.5 seconds
void showStatus(const char *text) {
  lcd.setCursor(0, 0);
  lcd.print(text);
  for (int i = strlen(text); i < 16; i++) lcd.print(' ');
  statusUntil = millis() + 1500;
}

void clearStatusIfDue() {
  if (statusUntil && millis() > statusUntil) {
    lcd.setCursor(0, 0);
    lcd.print("                ");
    statusUntil = 0;
  }
}

bool connectWiFi() {
  if (WiFi.status() == WL_CONNECTED) return true;

  Serial.print("Connecting to WiFi");
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  unsigned long t = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - t < 10000) {
    delay(500);
    Serial.print(".");
  }
  Serial.println();

  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("WiFi failed.");
    return false;
  }
  Serial.print("WiFi connected, IP: ");
  Serial.println(WiFi.localIP());
  return true;
}

bool connectMQTT() {
  if (client.connected()) return true;

  Serial.print("Connecting to MQTT...");
  String clientId = String("uno_r4_morse_") + BOARD_ID;
  // last will: if this board disappears, the broker publishes "offline"
  if (client.connect(clientId.c_str(), NULL, NULL, statusTopic.c_str(), 0, true, "offline")) {
    Serial.println("connected");
    client.publish(statusTopic.c_str(), "online", true);   // retained, so the game page sees it
    return true;
  }
  Serial.print("failed, rc=");
  Serial.println(client.state());
  return false;
}

// Connects to WiFi and MQTT if needed. Returns true if ready to send.
bool connectNetwork() {
  return connectWiFi() && connectMQTT();
}

void sendWord() {
  if (message.length() == 0) {
    Serial.println("Nothing to send.");
    showStatus("Nothing to send");
    return;
  }

  showStatus("Sending...");
  if (!connectNetwork()) {
    showStatus("Send failed");
    return;
  }

  if (client.publish(wordTopic.c_str(), message.c_str())) {
    Serial.print("Sent to ");
    Serial.print(wordTopic);
    Serial.print(": ");
    Serial.println(message);
    message = "";                      // start a fresh word
    updateLCD();
    showStatus("Sent!");
  } else {
    Serial.println("Publish failed.");
    showStatus("Send failed");
  }
}

// Checks both buttons
void checkButtons() {
  if (pressedEvent(deleteBtn) && message.length() > 0) {
    message.remove(message.length() - 1);
    Serial.print("Deleted. Text is now: ");
    Serial.println(message);
    updateLCD();
  }
  if (pressedEvent(sendBtn)) {
    sendWord();
  }
}

// Reads the knob and turns the mic ON/OFF
void checkMicSwitch() {
  analogRead(potPin);                  // dummy read lets the ADC settle
  int v = analogRead(potPin);

  if (!micOn && v > potOn) {
    micOn = true;
    Serial.println("Mic ON");
  } else if (micOn && v < potOff) {
    micOn = false;
    inSound = false;
    code = "";
    wordSpaceSent = true;
    digitalWrite(ledPin, LOW);
    Serial.println("Mic OFF");
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(ledPin, OUTPUT);
  pinMode(deletePin, INPUT_PULLUP);
  pinMode(sendPin, INPUT_PULLUP);
  analogReadResolution(10);

  lcd.init();
  lcd.backlight();
  lcd.clear();

  wordTopic   = String("pc/morse/") + BOARD_ID + "/word";
  statusTopic = String("pc/morse/") + BOARD_ID + "/status";
  client.setServer(MQTT_BROKER, MQTT_PORT);

  lcd.setCursor(0, 0);
  lcd.print("Connecting...");
  if (connectNetwork()) showStatus("Online");
  else showStatus("Offline");

  micOn = true;                        // start ON, then let the knob decide
  checkMicSwitch();

  Serial.print("Board ID: ");
  Serial.println(BOARD_ID);
  Serial.println("Ready. Short sound = dot, long sound = dash.");
  Serial.println("Button 3 = delete, button 4 = send, knob = mic on/off.");
}

void loop() {
  client.loop();                       // keeps the MQTT connection alive
  clearStatusIfDue();
  checkMicSwitch();

  // Mic is off: ignore sound, but the buttons still work
  if (!micOn) {
    checkButtons();
    delay(10);
    return;
  }

  // Measure loudness over 50 ms (buttons are checked during this too)
  unsigned long start = millis();
  int signalMax = 0, signalMin = 1023;
  while (millis() - start < sampleWindow) {
    int s = analogRead(micPin);
    if (s > signalMax) signalMax = s;
    if (s < signalMin) signalMin = s;
    checkButtons();
  }
  int level = signalMax - signalMin;
  if (debug) Serial.println(level);

  bool loud = level > threshold;
  unsigned long now = millis();
  digitalWrite(ledPin, loud ? HIGH : LOW);

  if (loud) {
    if (!inSound) {
      inSound = true;
      soundStart = now - sampleWindow;
    }
    lastLoud = now;
  }

  // Sound ended -> decide dot or dash
  if (inSound && now - lastLoud > soundEndGap) {
    unsigned long duration = lastLoud - soundStart;
    char symbol = (duration < dashTime) ? '.' : '-';
    if (code.length() < 7) code += symbol;
    Serial.print(symbol);
    Serial.print("  (");
    Serial.print(duration);
    Serial.println(" ms)");
    inSound = false;
    lastEnd = lastLoud;
    wordSpaceSent = false;
  }

  // Quiet long enough -> decode the letter and show it
  if (!inSound && code.length() > 0 && now - lastEnd > letterGap) {
    char letter = decodeMorse(code);
    Serial.print(code);
    Serial.print("  ->  ");
    Serial.println(letter);
    addToMessage(letter);
    code = "";
    updateLCD();
  }

  // Quiet much longer -> word space (Serial Monitor only)
  if (!inSound && code.length() == 0 && !wordSpaceSent && now - lastEnd > wordGap) {
    Serial.println("[space]");
    wordSpaceSent = true;
  }
}

const players = {};   // board id -> { score, tries, online }
let current = "", playing = false, locked = false;

function player(id) { return players[id] || (players[id] = { score: 0, tries: 0, online: false }); }

function renderPlayers() {
  const box = $("players");
  box.innerHTML = "";
  const ids = Object.keys(players).sort((a, b) => players[b].score - players[a].score || a.localeCompare(b));
  if (!ids.length) {
    const e = document.createElement("div");
    e.className = "empty"; e.textContent = "No boards yet";
    box.appendChild(e);
  }
  ids.forEach(id => {
    const row = document.createElement("div");
    row.className = "pl" + (players[id].online ? " on" : "");
    const n = document.createElement("span");
    n.className = "pn"; n.textContent = id;
    const s = document.createElement("b");
    s.textContent = players[id].score;
    row.append(n, s);
    box.appendChild(row);
  });
  const online = ids.filter(i => players[i].online).length;
  $("bTxt").textContent = online + (online === 1 ? " board" : " boards");
  pill("pBoard", online ? "on" : "");
}

// ---------- game ----------
function nextWord() {
  let w;
  do { w = WORDS[Math.floor(Math.random() * WORDS.length)]; } while (w === current && WORDS.length > 1);
  current = w;
  const box = $("word");
  box.className = "word";
  box.innerHTML = "";
  for (const ch of w) {
    const t = document.createElement("div");
    t.className = "tile";
    const c = document.createElement("div");
    c.className = "ch"; c.textContent = ch;
    const m = document.createElement("div");
    m.className = "m";
    for (const s of MORSE[ch]) {
      const i = document.createElement("i");
      i.className = s === "." ? "d" : "l";
      m.appendChild(i);
    }
    t.append(c, m);
    box.appendChild(t);
  }
  setFb("", "");
  locked = false;
}

function setFb(text, cls) { const f = $("fb"); f.textContent = text; f.className = "fb " + cls; }

function startGame() {
  if (playing) return;
  playing = true;
  $("waiting").classList.add("hidden");
  $("game").classList.remove("hidden");
  nextWord();
}

function judge(id, sent) {
  if (!playing || locked) return;
  sent = sent.trim().toUpperCase();
  if (!sent) return;
  const p = player(id);
  p.tries++;
  if (sent === current) {
    p.score++;
    renderPlayers();
    locked = true;                        // first board to get it wins the point
    $("word").className = "word ok";
    setFb(id + " got it! +1 point", "ok");
    setTimeout(nextWord, 1800);
  } else {
    const box = $("word");
    box.className = "word";
    void box.offsetWidth;                 // restart the shake animation
    box.className = "word bad";
    setFb(id + " sent \u201C" + sent + "\u201D. Try again.", "bad");
  }
}

$("startBtn").onclick = startGame;
$("skipBtn").onclick = () => { if (playing && !locked) nextWord(); };
$("hintBox").onchange = e => $("word").parentElement.classList.toggle("hints-off", !e.target.checked);

// ---------- connection pills ----------
function pill(id, state) { $(id).className = "pill " + state; }

// ---------- MQTT ----------
function setNet(t) { $("net").textContent = t; }
if (typeof mqtt === "undefined") {
  setNet("Couldn\u2019t load the MQTT library. Check your internet connection, then reload the page.");
  throw new Error("mqtt.js did not load");
}
let client = null;

function connectMqtt() {
  if (client) return;
  setNet("Connecting to the server\u2026");
  $("connBtn").textContent = "Disconnect";
  pill("pBroker", "wait");
  client = mqtt.connect(CFG.url, {
    clientId: "morse-game-" + Math.random().toString(16).slice(2, 8),
    reconnectPeriod: 3000
  });
  attach(client);
}

function disconnectMqtt() {
  if (!client) return;
  client.end();
  client = null;
  pill("pBroker", "");
  Object.values(players).forEach(p => p.online = false);
  renderPlayers();
  $("connBtn").textContent = "Connect";
  setNet("Disconnected. Press Connect to join the server.");
}

$("connBtn").onclick = () => { if (client) disconnectMqtt(); else connectMqtt(); };
renderPlayers();

function attach(client) {

client.on("connect", () => {
  pill("pBroker", "on");
  setNet("Connected to the server. Waiting for the board\u2026");
  client.subscribe([CFG.base + "/+/word", CFG.base + "/+/status"]);
});
client.on("reconnect", () => { pill("pBroker", "wait"); setNet("Reconnecting to the server\u2026"); });
client.on("close", () => pill("pBroker", ""));
client.on("error", e => setNet("Can\u2019t reach the server (" + e.message + "). Your network may be blocking port 8884."));

client.on("message", (topic, payload) => {
  const msg = payload.toString().trim();
  const parts = topic.split("/");         // e.g. pc/morse/board1/word
  const kind = parts[parts.length - 1];
  const id = parts[parts.length - 2];
  const p = player(id);
  if (kind === "status") {
    p.online = msg.toLowerCase() === "online";
    renderPlayers();
    if (p.online) startGame();
  } else if (kind === "word") {
    if (!p.online) { p.online = true; renderPlayers(); }
    if (!playing) startGame();            // a word arrived, so the board is clearly there
    else judge(id, msg);
  }
});
}
